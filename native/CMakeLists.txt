cmake_minimum_required(VERSION 3.16)
project(tensorrt_worker LANGUAGES CXX)
find_package(OpenCV REQUIRED)
add_executable(tensorrt_worker tensorrt_worker.cpp)
target_sources(tensorrt_worker PRIVATE runner.cpp)
target_compile_features(tensorrt_worker PRIVATE cxx_std_17)
target_include_directories(tensorrt_worker PRIVATE /usr/local/cuda-13.2/targets/sbsa-linux/include)
target_link_directories(tensorrt_worker PRIVATE /usr/local/cuda-13.2/targets/sbsa-linux/lib)
target_link_libraries(tensorrt_worker PRIVATE ${OpenCV_LIBS} nvinfer cudart)
